```javascript
function processTemplate(template, wizardData) {
  let processed = JSON.stringify(template);
  
  // Basic replacements
  processed = processed.replace(/{{SHOP_NAME}}/g, wizardData.shopName);
  processed = processed.replace(/{{PHONE}}/g, wizardData.phone);
  processed = processed.replace(/{{BOOKING_URL}}/g, wizardData.bookingUrl);
  
  // Address replacements
  const fullAddress = `${wizardData.street}, ${wizardData.city}, ${wizardData.state} ${wizardData.zip}`;
  processed = processed.replace(/{{FULL_ADDRESS}}/g, fullAddress);
  processed = processed.replace(/{{ENCODED_ADDRESS}}/g, encodeURIComponent(fullAddress));
  
  // Generated content
  processed = processed.replace(/{{HOURS_SUMMARY}}/g, generateHoursSummary(wizardData.hours));
  processed = processed.replace(/{{PRICING_SUMMARY}}/g, generatePricingSummary(wizardData.services));
  processed = processed.replace(/{{BARBER_LIST}}/g, generateBarberList(wizardData.barbers));
  
  // ... continue for all variables
  
  return JSON.parse(processed);
}
```

**Variables to Support:**

- Basic: SHOP_NAME, PHONE, EMAIL, WEBSITE
- Address: FULL_ADDRESS, ENCODED_ADDRESS, APPLE_MAPS_LINK, GOOGLE_MAPS_LINK
- Hours: HOURS_SUMMARY, MONDAY_HOURS, TUESDAY_HOURS, etc.
- Services: PRICING_SUMMARY, PRICING_MENU_TEXT, individual service prices
- Barbers: BARBER_LIST, BARBER_INFO_TEXT, specialist names
- Policies: CANCELLATION_POLICY, LATE_POLICY

### **2. Intent Detection System**

**Status:** [ ] Not Started [ ] In Progress [ ] Complete

**What’s Needed:**
Detect booking intent vs info-only questions

**Two Approaches:**

**A) Keyword-Based (Simpler):**

```javascript
function detectIntent(message) {
  const bookingKeywords = ['book', 'appointment', 'schedule', 'available', 'openings', 
                           'today', 'tomorrow', 'walk in', 'need a cut', 'need a fade'];
  
  const lowerMessage = message.toLowerCase();
  const hasBookingIntent = bookingKeywords.some(keyword => lowerMessage.includes(keyword));
  
  return hasBookingIntent ? 'booking' : 'info';
}
```

**B) AI-Based (Better):**

```javascript
async function detectIntent(message, conversationHistory) {
  const prompt = `
    Analyze this customer message and determine intent:
    - "booking": wants to schedule/book an appointment
    - "info": just asking for information (price, hours, location)
    
    Message: "${message}"
    
    Respond with just the intent type.
  `;
  
  const intent = await callAI(prompt);
  return intent.toLowerCase().trim();
}
```

### **3. Session State Management**

**Status:** [ ] Not Started [ ] In Progress [ ] Complete

**What’s Needed:**
Track conversation state to prevent spam

**Implementation:**

```javascript
class ConversationSession {
  constructor(userId) {
    this.userId = userId;
    this.leadCaptureInProgress = false;
    this.leadCaptureCompleted = false;
    this.leadCaptureStartedAt = null;
    this.capturedData = {};
    this.currentField = null;
  }
  
  startLeadCapture() {
    if (this.leadCaptureCompleted) {
      // Don't restart unless user explicitly says "book another"
      return false;
    }
    
    if (this.leadCaptureInProgress) {
      // Already in progress, continue existing flow
      return false;
    }
    
    this.leadCaptureInProgress = true;
    this.leadCaptureStartedAt = Date.now();
    this.currentField = 'serviceRequested';
    return true;
  }
  
  captureField(fieldName, value) {
    this.capturedData[fieldName] = value;
    
    // Determine next field
    const fields = ['serviceRequested', 'preferredDay', 'preferredTimeWindow', 
                    'name', 'phone', 'email', 'notes'];
    const currentIndex = fields.indexOf(fieldName);
    
    if (currentIndex < fields.length - 1) {
      this.currentField = fields[currentIndex + 1];
    } else {
      this.completeLeadCapture();
    }
  }
  
  completeLeadCapture() {
    this.leadCaptureInProgress = false;
    this.leadCaptureCompleted = true;
    this.currentField = null;
  }
  
  shouldTriggerCapture() {
    const cooldownMinutes = 5;
    if (this.leadCaptureCompleted && this.leadCaptureStartedAt) {
      const minutesSince = (Date.now() - this.leadCaptureStartedAt) / 1000 / 60;
      if (minutesSince < cooldownMinutes) {
        return false;
      }
    }
    return !this.leadCaptureInProgress && !this.leadCaptureCompleted;
  }
}
```

### **4. After-Hours Detection**

**Status:** [ ] Not Started [ ] In Progress [ ] Complete

**What’s Needed:**
Check if current time is within business hours

**Implementation:**

```javascript
function isAfterHours(schedule, timezone) {
  const now = new Date().toLocaleString('en-US', { timeZone: timezone });
  const currentTime = new Date(now);
  const day = currentTime.toLocaleDateString('en-US', { weekday: 'lowercase', timeZone: timezone });
  
  const todaySchedule = schedule[day];
  
  if (!todaySchedule || todaySchedule === null) {
    // Closed today
    return true;
  }
  
  const currentHour = currentTime.getHours();
  const currentMinute = currentTime.getMinutes();
  const currentTotalMinutes = currentHour * 60 + currentMinute;
  
  const [openHour, openMin] = todaySchedule.open.split(':').map(Number);
  const [closeHour, closeMin] = todaySchedule.close.split(':').map(Number);
  
  const openTotalMinutes = openHour * 60 + openMin;
  const closeTotalMinutes = closeHour * 60 + closeMin;
  
  return currentTotalMinutes < openTotalMinutes || currentTotalMinutes >= closeTotalMinutes;
}
```

**Status:** [ ] Not Started [ ] In Progress [ ] Complete

```javascript
async function runTestScenario(botId, scenario) {
  const conversation = new Conversation(botId);
  
  const response = await conversation.sendMessage(scenario.userMessage);
  
  const checks = {
    intentDetected: scenario.expectedIntent === response.detectedIntent,
    leadCaptureTriggered: scenario.expectLeadCapture === response.leadCaptureStarted,
    responseContains: scenario.expectedKeywords.every(kw => 
      response.text.toLowerCase().includes(kw)
    )
  };
  
  return {
    scenario: scenario.name,
    passed: Object.values(checks).every(check => check === true),
    checks
  };
}
```
